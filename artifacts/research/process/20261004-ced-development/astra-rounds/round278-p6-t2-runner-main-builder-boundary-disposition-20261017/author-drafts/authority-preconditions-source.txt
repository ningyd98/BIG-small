
def command_paths(command):
    directory = Path(command["output_directory"])
    expected = ["invocation.json", "stdout.txt", "stderr.txt", "junit.xml", "result.json",
                "actual-tool-return.json", "before/source-index.json", "source-after.json",
                "cases.json", "resource-summary.json", "persistent-file-database-index.json",
                "prerequisite.json"]
    if command["kind"] == "diagnostic_only":
        expected += ["passive/events.json", "passive/summary.json"]
    require(command["outputs"] == expected, "exact R276 finite output names required")
    return {"directory": directory, "invocation": directory / expected[0],
            "stdout": directory / expected[1], "stderr": directory / expected[2],
            "JUnit": directory / expected[3], "result": directory / expected[4],
            "actual_tool_return": directory / expected[5], "before": directory / expected[6],
            "after": directory / expected[7], "cases": directory / expected[8],
            "resources": directory / expected[9], "index": directory / expected[10],
            "prerequisite": directory / expected[11], "passive": directory / "passive"}


def verify_db_pairs(rows, expected_count):
    require(type(rows) is list and len(rows) == expected_count, "exact retained DB pair count differs")
    present, absent_count = 0, 0
    for row in rows:
        require(checked(row["pin"]) == checked(row["archive"]), "retained DB own original/archive differs")
        require(type(row["sidecars"]) is list and len(row["sidecars"]) == 3, "three real DB sidecar states required")
        for side in row["sidecars"]:
            require(side["is_symlink"] is False, "old DB sidecar cannot become symlink")
            if side["exists"]:
                require(checked(side["pin"]) == checked(side["archive"]), "present DB sidecar pair differs")
                present += 1
            else:
                require(side["pin"] is None, "missing sidecar cannot invent a current pin")
                absence(side["path"])
                absent_count += 1
    return {"present": present, "absent": absent_count}


def authority(context):
    plan, gate = context["plan"], context["source_gate"]
    changes = context["context_manifest"]["changes"]
    expected_context = {r["path"] for r in plan["context_update"]["before_inputs"]}
    require(len(changes) == 2, "two precise context generations required")
    mapped, context_after = {}, []
    for row in changes:
        original = pin_identity(row["before"]["original"])
        raw = bytes_record(row["before"])
        require(original["path"] in expected_context and original["path"] not in mapped
                and row["after"]["path"] == original["path"], "unexpected context generation")
        require(any(pin_identity(q) == original for q in plan["context_update"]["before_inputs"]),
                "context original is not the frozen planning identity")
        after = checked(row["after"])
        require(checked(row["after_archive"]) == after
                and Path(row["after"]["path"]).lstat().st_mode == row["mode"],
                "context after/full prepared archive/mode differs")
        mapped[original["path"]] = (original, raw)
        context_after.append(pin_identity(row["after"]))
    require(set(mapped) == expected_context, "context mapping incomplete")
    require(len(plan["inputs"]) == 50
            and [pin_identity(x) for x in context["observation"]["inputs"]]
            == [pin_identity(x) for x in plan["inputs"]], "exact50 initial authorities required")
    records, mapped_count = [], 0
    for ordinal, row in enumerate(plan["inputs"]):
        item = pin_identity(row)
        if item["path"] in mapped:
            original, raw = mapped[item["path"]]
            require(item == original and bytes_pin(item["path"], raw) == item,
                    "old context identity has no exact physical generation")
            resolution = "EXACT_CONTEXT_BEFORE_TO_FROZEN_AFTER"
            mapped_count += 1
        else:
            checked(item)
            resolution = "CURRENT_EXACT"
        records.append({"ordinal": ordinal, "pin": item, "resolution": resolution})
    require(mapped_count == 2, "only two old planning context identities may map")
    for row in context["supplement"]["inputs"]:
        checked(row)
    for row in gate["fixed_inputs"] + context["stable_controls"]:
        checked(row)
    index_pin = next(q for q in plan["inputs"]
                     if q["path"] == str(PARENT / "implementation/dual_progress/before/source-index.json"))
    old_index = decode_json(checked(index_pin))
    prior = old_index["authority_before"]
    require(len(old_index["sources"]) == 27 and len(prior["protected"]) == 211
            and len(prior["accepted_history_generations"]) == 68,
            "accepted R268 finite27/211/68 authority differs")
    manifest_pin = next(q for q in plan["inputs"] if q["path"] == str(R274 / "before/manifest.json"))
    manifest = decode_json(checked(manifest_pin))
    test_before = bytes_record(manifest)
    after_pin = next(q for q in plan["inputs"] if q["path"] == str(R274 / "implementation/after-test.py"))
    test_after = checked(after_pin)
    old_plan_pin = next(q for q in plan["inputs"] if q["path"] == str(R274 / "plan.json"))
    old_plan = decode_json(checked(old_plan_pin))
    expected = test_before
    for change in old_plan["exact_source_change"]["changes"]:
        old = change["old"].encode()
        require(expected.count(old) == 1, "R274 unique original substitution differs")
        expected = expected.replace(old, change["new"].encode(), 1)
    require(expected == test_after and test_before[:168003] == test_after[:168003]
            and hashlib.sha256(test_after).hexdigest() == TEST_SHA,
            "actual R274 five-edit generation differs")
    acceptance_pin = context["supplement"]["inputs"][0]
    require(acceptance_pin["path"] == str(R274 / "root-independent-source-acceptance.json"),
            "real immutable R274 independent acceptance required")
    checked(acceptance_pin)
    sources = []
    for row in old_index["sources"]:
        original = pin_identity(row)
        if original["path"] == manifest["original"]["path"]:
            require(original == pin_identity(manifest["original"])
                    and checked(row["physical"]) == test_before
                    and base64.b64decode(row["base64"], validate=True) == test_before,
                    "old test must resolve only through complete R274 physical before")
            current = bytes_pin(original["path"], test_after)
            require(checked(current) == test_after, "live R274 test differs")
            physical = after_pin
        else:
            raw = checked(original)
            require(raw == checked(row["physical"]) == base64.b64decode(row["base64"], validate=True),
                    "one of26 other complete source generations differs")
            current, physical = original, row["physical"]
        require(Path(current["path"]).lstat().st_mode == row["mode"], "current source mode differs")
        sources.append({**current, "mode": row["mode"], "physical": physical})
    for row in prior["protected"]:
        checked(row)
    missing, sides = prior["missing_registry35"], prior["old_absent_sidecars90"]
    require(len(missing) == 35 and len(sides) == 90, "old35/90 boundaries differ")
    for row in missing:
        require(row["disposition"] == "HISTORICAL_BYTES_CURRENTLY_UNAVAILABLE_R245"
                and row["current_bytes"] is None and row["current_sha256"] is None
                and row["hash_match"] is None, "historical evidence loss cannot become recovered bytes")
        absence(row["original"]["path"])
    for row in sides:
        require(row["exists"] is False and row["is_symlink"] is False, "old sidecar state differs")
        absence(row["path"])
    groups = prior["old_retained_groups"]
    require(type(groups) is dict and set(groups) == {"R242_R248", "R251"}, "exact old DB groups required")
    for key, count in (("R242_R248", 12), ("R251", 18)):
        require(type(groups[key]) is list and len(groups[key]) == count, "old retained DB count differs")
        for row in groups[key]:
            checked(row["pin"])
            require(len(row["sidecars"]) == 3, "three old retained sidecars required")
            for side in row["sidecars"]:
                require(side["exists"] is False and side["is_symlink"] is False and side["pin"] is None,
                        "old retained DB sidecar absence differs")
                absence(side["path"])
    original_pairs = prior["R258_retained_DB_pairs"]
    require(verify_db_pairs(original_pairs, 10) == {"present": 2, "absent": 28},
            "R258 exact present/absent sidecar boundary differs")
    added_pairs, old_root_maps = {}, []
    for key, count, acceptance_name in (
        ("wait_memory_failure", 4, "root-wait-memory-independent-acceptance.json"),
        ("dual_progress", 2, "root-dual-independent-acceptance.json"),
    ):
        rp = next(q for q in plan["inputs"] if q["path"] == str(PARENT / "implementation" / key / "result.json"))
        result = decode_json(checked(rp))
        pairs = result["new_retained_databases"]
        verify_db_pairs(pairs, count)
        added_pairs[key] = pairs
        ap = next(q for q in plan["inputs"] if q["path"] == str(PARENT / acceptance_name))
        accepted = decode_json(checked(ap))
        bytes_record(accepted["root_before"])
        old_root_maps.append({"acceptance": ap, "generation": accepted["root_before"]})
    # These six finite physical controls explain old generations; no old logical
    # path is waived or demanded to match its changed live role/context file.
    controls = []
    for row in old_index["control_before"]:
        bytes_record(row)
        if row["original"]["path"] == str(PARENT / "root-command-ledger.json"):
            require(pin_identity(row["original"]) == pin_identity(old_root_maps[1]["generation"]["original"]),
                    "old dual ROOT control must join its actual independent settlement before")
        controls.append(row)
    for role in (str(PARENT / "root-command-ledger.json"), str(PARENT / "implementation/command-ledger.json")):
        q = next(q for q in plan["inputs"] if q["path"] == role)
        value = decode_json(checked(q))
        require(set(value["commands"]) == {"wait_memory_failure", "dual_progress"}
                and all(x["used"] == 1 and x["remaining"] == 0 and x["state"].startswith("SPENT")
                        for x in value["commands"].values()), "R268 roles must remain spent")
    for row in context["sources"].values():
        checked(row)
    return {"input_records": records, "supplement_inputs": context["supplement"]["inputs"],
            "fixed_inputs": gate["fixed_inputs"], "context_after": context_after,
            "sources": sources, "protected": prior["protected"], "source_authority": index_pin,
            "accepted_history_generations": prior["accepted_history_generations"],
            "history_reaudited_as_current": False, "old_control_generations": controls,
            "old_ROOT_settlement_generations": old_root_maps,
            "test_generation": {"before": manifest, "after": after_pin, "ROOT_acceptance": acceptance_pin},
            "missing_registry35": missing, "old_absent_sidecars90": sides,
            "old_retained_groups": groups, "R258_retained_DB_pairs": original_pairs,
            "R268_retained_DB_pairs": added_pairs}


def preconditions():
    require(len(sys.argv) == 2, "one R276 key required")
    name = sys.argv[1]
    require(name in {"wait_memory", "wait_sqlite", "dual_first_invalidation"}
            and Path.cwd() == ROOT and Path(sys.executable).absolute() == ROOT / ".venv/bin/python",
            "exact R276 command/interpreter/cwd required")
    plan_raw, obs_raw = stable_bytes(D / "plan.json"), stable_bytes(D / "input-observation.json")
    supplement_raw = stable_bytes(D / "prerequisite-supplement.json")
    require(hashlib.sha256(plan_raw).hexdigest() == PLAN_SHA
            and hashlib.sha256(obs_raw).hexdigest() == OBS_SHA
            and hashlib.sha256(supplement_raw).hexdigest() == SUPPLEMENT_SHA, "frozen R276 authority differs")
    plan, observation, supplement = map(decode_json, (plan_raw, obs_raw, supplement_raw))
    command = plan["commands"][name]
    require(command["limits"]["child_seconds"] == 600 and command["limits"]["cleanup_total_seconds"] == 15
            and command["limits"]["raw_stream_threshold_bytes_each"] == 33554432
            and command["node_count"] == 1
            and [a for a in command["child_argv"] if "::" in a] == [command["node"]],
            "exact R276 one-node finite command required")
    require(command["child_env"]["BIGSMALL_R158_COMMAND"] == "t2_green",
            "frozen resource fixture command name required")
    trace_requested = "r276_session_evidence" in command["child_argv"]
    require(trace_requested == (name == "dual_first_invalidation")
            and "r268_failure_evidence" not in command["child_argv"], "wait cannot load a trace observer")
    source_path, script_path = D / "source-gate.json", D / "script-gate.json"
    gate, script_gate = read_json(source_path), read_json(script_path)
    require(gate["round"] == script_gate["round"] == "R276"
            and gate["status"] == "VERIFIED_SOURCE_ONLY" and script_gate["status"] == "VERIFIED_BY_ROOT",
            "ROOT complete source/runtime gates required")
    require(pin_identity(gate["plan"]) == bytes_pin(D / "plan.json", plan_raw)
            and pin_identity(gate["observation"]) == bytes_pin(D / "input-observation.json", obs_raw)
            and pin_identity(gate["supplement"]) == bytes_pin(D / "prerequisite-supplement.json", supplement_raw)
            and pin_identity(script_gate["plan"]) == pin_identity(gate["plan"])
            and pin_identity(script_gate["source_gate"]) == _t2_pin(source_path), "ROOT authority generation differs")
    sources = gate["sources"]
    require(set(sources) == {"plugin", "runner"} and script_gate["sources"] == sources
            and pin_identity(sources["runner"]) == _t2_pin(Path(__file__)), "two exact reviewed R276 sources required")
    roles = {key: plan["runner_contract"]["roles"][key] for key in ("ROOT", "AUTHOR")}
    require(script_gate["commands"] == plan["commands"] and script_gate["role_files"] == roles,
            "ROOT commands/role-file interface differs")
    for row in (gate["input_check"], gate["native"], gate["supplement"]):
        checked(row)
    require(type(gate["fixed_inputs"]) is list
            and all(q["path"] not in set(roles.values()) | {str(source_path), str(script_path)}
                    for q in gate["fixed_inputs"]), "no live mutable control in fixed input loop")
    context_manifest = decode_json(checked(gate["context_manifest"]))
    require(context_manifest["round"] == "R276" and context_manifest["T2_complete"] is False,
            "exact ROOT factual context map required")
    reservation_dir = D / "root-reservations"
    require(script_gate["reservation_directory"] == str(reservation_dir), "exact reservation directory required")
    reservation_path = reservation_dir / (name + ".json")
    reservation = read_json(reservation_path)
    require(reservation["round"] == "R276" and reservation["command_id"] == name
            and type(reservation["reservation_id"]) is str and reservation["reservation_id"]
            and reservation["command"] == command and reservation["sources"] == sources
            and pin_identity(reservation["plan"]) == pin_identity(gate["plan"])
            and pin_identity(reservation["source_gate"]) == _t2_pin(source_path)
            and pin_identity(reservation["script_gate"]) == _t2_pin(script_path)
            and pin_identity(reservation["context_manifest"]) == pin_identity(gate["context_manifest"]),
            "exact selected ROOT reservation required")
    require(len(reservation["role_before"]) == 2, "two full role-before generations required")
    full_before = {row["original"]["path"]: bytes_record(row) for row in reservation["role_before"]}
    require(set(full_before) == set(roles.values()), "exact new role-before paths required")
    root_before, author_before = (decode_json(full_before[roles[key]]) for key in ("ROOT", "AUTHOR"))
    for role, value in (("ROOT", root_before), ("AUTHOR", author_before)):
        require(value["round"] == "R276" and value["role"] == role
                and pin_identity(value["plan"]) == pin_identity(gate["plan"])
                and set(value["commands"]) == set(plan["commands"]), "new role identity differs")
        entry = value["commands"][name]
        require(entry["authorized"] == 1 and entry["used"] == 0 and entry["remaining"] == 1
                and entry["state"] == "HELD_PENDING_ROOT_SOURCE_INPUT_ARGV_GATE"
                and entry["command"] == command, "selected new role must be HELD0/1")
    continuation = {"reservation_id": reservation["reservation_id"], "plan": reservation["plan"],
                    "source_gate": reservation["source_gate"], "script_gate": reservation["script_gate"],
                    "sources": sources, "command": command}
    root_expected = copy.deepcopy(root_before)
    selected = root_expected["commands"][name]
    selected.update(state="RESERVED_BY_ROOT", used=1, remaining=0, reservation_id=reservation["reservation_id"])
    selected.setdefault("continuation", {})["R276"] = continuation
    require(reservation["root_role_after"]["path"] == roles["ROOT"]
            and decode_json(checked(reservation["root_role_after"])) == root_expected,
            "ROOT selected-key CAS transition differs")
    author_raw = checked(reservation["author_role_before"])
    require(reservation["author_role_before"]["path"] == roles["AUTHOR"]
            and author_raw == full_before[roles["AUTHOR"]], "author exact current before differs")
    stable = [_t2_pin(source_path), _t2_pin(script_path), _t2_pin(reservation_path),
              reservation["root_role_after"], gate["context_manifest"]]
    context = {"plan": plan, "observation": observation, "supplement": supplement, "command": command,
               "name": name, "source_gate": gate, "sources": sources, "reservation": reservation,
               "context_manifest": context_manifest, "stable_controls": stable,
               "author_before": author_before, "author_raw": author_raw,
               "author_path": Path(roles["AUTHOR"]), "continuation": continuation}
    context["paths"] = command_paths(command)
    require(context["paths"]["directory"] == D / "implementation" / name, "exact new output directory required")
    absence(context["paths"]["directory"])
    context["before_check"] = authority(context)
    return context


def save_before(context):
    paths = context["paths"]
    output = paths["directory"]
    output.mkdir(exist_ok=False)
    (output / "resources").mkdir(exist_ok=False)
    if context["name"] == "dual_first_invalidation":
        paths["passive"].mkdir(exist_ok=False)
    sources = []
    for row in context["before_check"]["sources"]:
        raw = checked(row)
        archive = write_bytes(output / "before/sources" / Path(row["path"]).relative_to(ROOT), raw, row["mode"])
        sources.append({**row, "physical": archive, "base64": base64.b64encode(raw).decode("ascii")})
    controls = []
    for ordinal, row in enumerate(context["stable_controls"] + [bytes_pin(context["author_path"], context["author_raw"])]):
        raw = checked(row)
        archive = write_bytes(output / "before/controls" / (str(ordinal) + ".bin"), raw)
        controls.append({"original": row, "archive": archive, "base64": base64.b64encode(raw).decode("ascii")})
    return write_json(paths["before"], {"round": "R276", "command_id": context["name"],
        "sources": sources, "new_sources": context["sources"], "control_before": controls,
        "authority_before": context["before_check"], "runtime_scope": context["command"]["kind"]})
