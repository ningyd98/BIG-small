def encoded(value):
    return (json.dumps(value,ensure_ascii=False,indent=2,allow_nan=False)+'\n').encode()
def identity(row):
    return {k:row[k] for k in ('path','bytes','sha256')}
def key(row):
    return tuple(row[k] for k in ('path','bytes','sha256'))
def pin(path,raw=None):
    raw = path.read_bytes() if raw is None else raw
    return {'path':str(path),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
def fixed(row):
    path = Path(row['path'])
    raw = path.read_bytes()
    if path.is_symlink() or pin(path,raw)!=identity(row):
        raise ValueError('fixed input differs: '+str(path))
    return raw
def save(path,value):
    path.parent.mkdir(parents=True,exist_ok=True)
    raw = value if isinstance(value,bytes) else encoded(value)
    with path.open('xb') as handle:
        if handle.write(raw)!=len(raw):
            raise OSError('partial exclusive metadata write')
    if path.read_bytes()!=raw:
        raise ValueError('exclusive metadata readback differs')
    return pin(path,raw)
def read(path):
    return json.loads(path.read_bytes())
def check(label,condition):
    checks.append({'label':label,'ok':bool(condition)})
def require_checks():
    if any(not x['ok'] for x in checks):
        raise ValueError('settlement integrity condition failed')
def actual_completion(directory,transport,expected_exit):
    actual = read(directory/'actual-tool-return.json')
    initial,polls,final = actual['initial'],actual['polls'],actual['final']
    check('actual_once_'+directory.name,actual['attempt']==1 and actual['retry']==0
          and final==actual['actual_tool_return']==(polls[-1] if polls else initial)
          and final['exit_code']==expected_exit and actual['invocation']==transport['invocation']
          and actual['independent_stdout'] is None and actual['independent_stderr'] is None)
    if polls:
        check('actual_same_session_'+directory.name,initial.get('session_id') is not None
              and all(x.get('session_id')==initial['session_id'] for x in polls[:-1])
              and len(actual['poll_invocations'])==len(polls)
              and all(x['session_id']==initial['session_id'] and x['chars']==''
                      and 0<x['yield_time_ms']<=60000 for x in actual['poll_invocations']))
    return actual
def close_controls(before,expected):
    for row in before:
        path = Path(row['path'])
        raw = base64.b64decode(row['base64'],validate=True)
        if pin(path,raw)!=identity(row):
            raise ValueError('complete original control bytes differ')
        actual = path.read_bytes()
        if actual!=expected.get(str(path),raw):
            raise ValueError('exact original control transition differs')

        print(json.dumps({'command':'root_settlement','exit_code':1,
                          'result_pin':pin(OUT/'result.json')},ensure_ascii=False))
        raise SystemExit(1)
