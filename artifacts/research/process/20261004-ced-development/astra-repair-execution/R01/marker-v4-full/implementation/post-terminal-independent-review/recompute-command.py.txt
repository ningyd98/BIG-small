def check(condition, reason, **details):
    if not condition:
        findings.append({"reason": reason, **details})

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def finite(value):
    return type(value) in {int, float} and math.isfinite(value)

def strict(raw):
    def pairs(values):
        result = {}
        for key, value in values:
            if key in result:
                raise ValueError("duplicate JSON key")
            result[key] = value
        return result
    def constant(value):
        raise ValueError("nonfinite JSON constant " + value)
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=constant)

def document(path):
    raw = path.read_bytes()
    pins[str(path.relative_to(root))] = {"bytes": len(raw), "sha256": digest(raw)}
    return strict(raw)

def lines(path):
    h, size = hashlib.sha256(), 0
    with path.open("rb") as stream:
        for line in stream:
            h.update(line)
            size += len(line)
            if not line.endswith(b"\n"):
                raise ValueError("truncated derived JSONL")
            yield strict(line)
    pins[str(path.relative_to(root))] = {"bytes": size, "sha256": h.hexdigest()}

def audit(event, args):
    if event != "open":
        return
    raw, mode, flags = args
    if isinstance(raw, (str, bytes, os.PathLike)):
        path = Path(os.fsdecode(raw)).resolve()
        writing = (isinstance(mode, str) and any(mark in mode for mark in "wax+")) or (isinstance(flags, int) and bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC)))
        if writing and not path.is_relative_to(out):
            raise RuntimeError("post-terminal reviewer writes outside owned review directory")
        if path.suffix == ".gz":
            raise RuntimeError("post-terminal reviewer must not read compressed raw/journals")
