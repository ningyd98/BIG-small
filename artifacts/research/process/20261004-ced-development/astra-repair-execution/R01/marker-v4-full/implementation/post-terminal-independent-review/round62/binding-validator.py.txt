def require(condition, reason):
    if not condition:
        raise ValueError(reason)

def hash_bytes(raw):
    return hashlib.sha256(raw).hexdigest()

def metadata_pin(path):
    require(path != DERIVED and path.suffix != ".gz", "forbidden stream/raw protection read")
    raw = path.read_bytes()
    return {"bytes": len(raw), "sha256": hash_bytes(raw)}

def write(name, value):
    with (OUT / name).open("x", encoding="utf-8") as stream:
        json.dump(value, stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write("\n")

def stat_stream():
    value = DERIVED.stat()
    return {"bytes": value.st_size, "mtime_ns": value.st_mtime_ns, "inode": value.st_ino, "device": value.st_dev}

def strict_json(raw):
    def pairs(items):
        result = {}
        for key, value in items:
            require(key not in result, "duplicate JSON field")
            result[key] = value
        return result
    def constant(value):
        raise ValueError("nonfinite JSON constant " + value)
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=constant)

def audit(event, args):
    if event == "open":
        raw, mode, flags = args
        if isinstance(raw, (str, bytes, os.PathLike)):
            path = Path(os.fsdecode(raw)).resolve()
            writing = (isinstance(mode, str) and any(mark in mode for mark in "wax+")) or (isinstance(flags, int) and bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC)))
            if writing:
                require(path.is_relative_to(OUT), "write outside exclusive round62 directory")
            else:
                require(path.suffix != ".gz", "raw or compressed journal read forbidden")
                if path.suffix == ".jsonl":
                    require(path in {DERIVED, OLD / "first-derived-line-original.jsonl"}, "old or unrelated derived stream read forbidden")
    if event == "import" and args and str(args[0]).startswith("cloud_edge_robot_arm"):
        raise RuntimeError("project imports forbidden")
